import flet as ft
import os
from langchain_community.document_loaders import TextLoader, PyPDFLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from langchain_huggingface import HuggingFaceEmbeddings
from langchain_openai import ChatOpenAI
from langchain_ollama import ChatOllama
from langchain.chains import RetrievalQA

# ====================== 全局配置 ======================
PERSIST_DIR = "./chroma_db"
# 本地嵌入模型，离线向量化文档
embedding_model = HuggingFaceEmbeddings(model_name="all-MiniLM-L6-v2")

vector_db = None
qa_chain = None

# ====================== RAG核心函数 ======================
def init_vector_db():
    """初始化向量库，存在则加载，不存在新建"""
    global vector_db
    vector_db = Chroma(
        persist_directory=PERSIST_DIR,
        embedding_function=embedding_model
    )
    return vector_db

def load_file_to_db(file_path):
    """加载PDF/TXT文档，切片存入向量库"""
    if not os.path.exists(file_path):
        raise Exception("文件不存在")
    # 文件加载器
    if file_path.lower().endswith(".pdf"):
        loader = PyPDFLoader(file_path)
    elif file_path.lower().endswith(".txt"):
        loader = TextLoader(file_path, encoding="utf-8")
    else:
        raise Exception("仅支持 .pdf 和 .txt 文件")
    docs = loader.load()
    # 文本分割器
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=800,
        chunk_overlap=100,
        length_function=len
    )
    chunks = splitter.split_documents(docs)
    vector_db.add_documents(chunks)
    vector_db.persist()
    return len(chunks)

def build_llm_chain(mode: str, api_key: str, model_name: str, base_url: str):
    """
    构建问答链
    :param mode: local 本地Ollama；api 外接OpenAI兼容API
    :param api_key: API密钥，本地模式可不填
    :param model_name: 模型名称，如 qwen2:7b / gpt-3.5-turbo
    :param base_url: API地址，本地模式忽略
    """
    global qa_chain
    retriever = vector_db.as_retriever(search_kwargs={"k": 3})
    if mode == "local":
        llm = ChatOllama(model=model_name)
    else:
        llm = ChatOpenAI(
            api_key=api_key,
            model=model_name,
            base_url=base_url if base_url else None
        )
    qa_chain = RetrievalQA.from_chain_type(
        llm=llm,
        chain_type="stuff",
        retriever=retriever,
        return_source_documents=True
    )
    return True

def rag_query(question: str):
    """执行RAG问答，返回答案+来源文档"""
    global qa_chain
    if qa_chain is None:
        raise Exception("请先配置模型并加载知识库")
    result = qa_chain.invoke({"query": question})
    answer = result["result"]
    source_docs = result["source_documents"]
    source_text = "\n=====参考文档片段=====\n"
    for idx, doc in enumerate(source_docs):
        source_text += f"{idx+1}. {doc.page_content[:300]}...\n"
    return answer + source_text

本地 RAG 桌面助手 项目开发文档
1 项目概述
1.1 项目简介
本项目为桌面端 RAG 知识库问答工具，基于 Python + Flet GUI 框架开发，集成 LangChain RAG 编排能力与 Chroma 嵌入式向量库。系统支持双模型运行模式：本地 Ollama 部署大模型、外接兼容 OpenAI 协议的 API 大模型，可导入 PDF、TXT 私有文档构建本地知识库，实现文档检索问答。所有文档向量数据可保存在本地，支持离线使用，可一键打包为 Windows/Mac 桌面程序。
1.2 技术栈选型
表格
模块	选型	说明
GUI 界面	Flet	纯 Python 编写界面，无需 HTML/JS 前端代码，跨平台打包，开发效率高
RAG 编排框架	LangChain	文档加载、文本切分、检索链管理，RAG 业务逻辑编排
向量数据库	Chroma	嵌入式向量库，无需独立服务进程，向量库随程序本地持久化存储
嵌入模型	all-MiniLM-L6-v2	HuggingFace 开源本地向量嵌入模型，文档向量化可离线，不需要联网
大模型支持	Ollama（本地）+ OpenAI 兼容 API（外接）	本地模式：Ollama 加载 GGUF 格式大模型（Qwen、Llama3 等）；外接模式：支持 DeepSeek、通义千问、智谱等 API 接口
文档解析	PyPDF + Python 内置 TextLoader	支持 PDF、TXT 格式文档读取
1.3 功能清单
知识库管理：选择本地 PDF/TXT 文件，上传并向量化存入本地向量库
模型切换：本地 Ollama 模型 / 外接 API 模型两种模式一键切换
模型配置：模型名称填写、API Key、自定义 BaseURL 配置（API 模式）
RAG 问答：输入问题，检索知识库返回答案，并附带参考文档片段
数据本地持久化：向量数据库保存在本地目录，重启程序知识库不丢失
桌面打包：Flet 打包命令，可编译为独立 exe 桌面程序交付使用
1.4 适用场景
企业私有文档、技术手册、学习资料本地问答，数据不上云
个人知识库，本地离线使用大模型 + 私有文档检索
快速定制轻量桌面 RAG 工具，无需 Web 服务部署
2 系统架构设计
2.1 整体架构分层
表现层（GUI）：Flet 桌面界面，提供可视化操作入口：文件选择、模型参数配置、问答输入输出展示
RAG 业务层（LangChain）：文档加载、文本分割、向量检索、问答链路组装
向量存储层（Chroma）：文档切片向量存储，持久化到本地文件夹
LLM 推理层
本地推理：Ollama 服务，本地运行 GGUF 大模型
API 推理：调用第三方大模型 API 服务，网络请求完成推理
嵌入层：本地 Embedding 模型，对文档和用户问题做向量转换，离线计算向量
2.2 核心 RAG 工作流程
用户选择文档，系统读取文档内容
文档内容进行文本切分，生成固定长度文本块，设置重叠窗口保证上下文连贯
本地 Embedding 模型将文本块转为向量，存入 Chroma 向量库并持久化
用户输入提问，问题转为向量，在向量库检索相似度最高的文本片段
将检索到的文档片段 + 用户问题一起送入大模型（本地 Ollama 或外接 API）
大模型基于参考文档生成回答，返回答案和对应的参考文档片段到界面展示

7 优缺点与优化方向
7.1 方案优势
Flet 开发速度快，不用前端技术栈，Python 全栈完成桌面 GUI

3 关键结论
单纯看前端界面美观度、功能丰富度：腾讯 WeKnora 强很多。它是完整产品化 Web 前端，布局、交互、可视化能力是生产级，开箱即用，不用自己写前端页面。
如果你想要独立桌面 exe 程序，而不是网页服务：WeKnora 不适合。WeKnora 只能浏览器访问，不能直接打包成桌面客户端；而 Flet 的核心优势就是做独立桌面程序。

Cherry Studio 技术栈文档
Cherry Studio：开源跨平台 AI 桌面客户端（Windows/macOS/Linux），Electron + React + TypeScript 开发，AGPL-3.0 开源协议。
1 核心技术栈
桌面外壳：Electron
用 Chromium 网页内核打包成桌面程序，本质是网页技术做桌面软件。
一套前端代码，打包输出 Windows、Mac、Linux 安装包。
前端 UI：React + TS（TypeScript）
页面、侧边栏、对话界面、知识库面板全部是 React 组件。
UI 交互、动画、多标签页、弹窗、流式对话渲染都是前端实现。
后端业务逻辑（客户端内）
Node.js，跑在 Electron 主进程，负责：
调用 LLM API / Ollama 本地模型通信
本地知识库 RAG 处理
文件解析、向量入库、数据库读写
模型管理、API 密钥管理、本地缓存
内置 RAG 知识库模块
向量库：Sqlite + 向量扩展（本地嵌入式，不需要独立服务）
Embedding：BGE 系列（bge-m3）本地嵌入模型
检索：混合检索（BM25 关键词检索 + 向量检索）+ Rerank 重排
文档解析：支持 PDF、docx、pptx、xlsx 等十余种文档，内置文档解析器
支持多知识库独立管理、批量上传、网页抓取、引用溯源
模型对接层
外接 API：OpenAI、DeepSeek、智谱、Anthropic 等几十家服务商
本地模型：对接 Ollama / LM Studio，本身不直接跑 GGUF 模型，只是客户端调用本地推理服务

Cherry Studio 界面好看的根本原因：React 成熟前端生态，可以直接使用大量 UI 组件库，做复杂布局、动画、弹窗、多标签、文档预览。

内置混合检索 + Rerank，多知识库，支持 Word/PPT/Excel/PDF 也能有这种功能吗
文档：Python+Flet 方案能否实现「混合检索 + Rerank + 多知识库 + Word/PPT/Excel/PDF」
结论先行
✅ 完全可以实现，但不是开箱自带，需要自己集成第三方 Python 库开发。
Cherry Studio 是内置做好了这一整套能力；而你的 Python+Flet 方案是基于 LangChain 生态组装，所有能力都可以补上，只是需要增加代码与依赖。
代价：代码量增加、项目复杂度上升、打包 exe 体积变大；Flet 只负责界面，检索 / 文档解析逻辑全部由 Python 后端实现。
一、分项能力可行性说明
1. 多知识库（多个独立知识库隔离）
可以实现
原理：Chroma/Qdrant 嵌入式向量库支持不同集合 (collection)，每个知识库对应一个独立 collection。
界面：Flet 增加下拉选择 / 列表，切换不同知识库
存储：每个知识库可以单独命名、独立向量空间，互不干扰
操作：新建知识库、删除知识库、查看每个库里面的文档列表
注意：原生 Chroma 的 Collection 多租户能力够用，适合单机桌面程序；大规模文档推荐嵌入式 Qdrant 替代 Chroma。
2. 混合检索：BM25 关键词检索 + 向量检索
可以实现
LangChain 生态可以集成：
BM25：rank_bm25 或者 elasticsearch（轻量单机优先 rank_bm25，不需要额外服务）
向量检索：Chroma/Qdrant 向量检索
融合策略：
分别拿到 BM25 结果列表、向量检索结果列表
对两份结果做结果融合（RRF 倒数排名融合 Reciprocal Rank Fusion），合并成一份候选文档列表
单机桌面程序首选：rank_bm25（纯内存，无服务，适合打包进 exe），不要用 Elasticsearch（太重，不适合桌面客户端）。
3. Rerank 重排
可以实现
Python 直接调用 BGE-reranker、bge-reranker-v2-m3 等开源重排模型。
本地离线 Rerank：bge-reranker，模型下载到本地，离线运行，不需要联网
也可以调用 API 版本 Rerank（联网）
流程：混合检索拿到候选文档 → Rerank 模型对【问题 + 候选片段】打分，重新排序，过滤低分片段，再送入 LLM。
这一步是提升 RAG 效果的关键，Cherry Studio 也是 BGE 系列 rerank。
4. 文档解析：Word / PPT / Excel / PDF / TXT
可以实现，Python 生态有成熟解析库
表格
文件类型	Python 库	备注
PDF	pypdf / pdfplumber	pdfplumber 解析表格效果更好
Word(.docx)	python-docx	读取文本，支持简单表格；不支持老版.doc
PPT(.pptx)	python-pptx	提取幻灯片文字
Excel(.xlsx)	openpyxl	读取单元格文本
TXT	内置	原生支持
限制：
仅支持新版 Office 格式（docx/pptx/xlsx），旧版.doc很难解析
图片里的文字（图片 PDF、图片 PPT）需要额外 OCR（paddleocr，增加打包体积）
复杂表格、排版、图表只能提取文字，无法保留原样式（和 CherryStudio 行为一致）
